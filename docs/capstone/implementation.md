```rust
pub enum QuarryError {
    #[error("{0}")] Refusal(String),   // exit 1
    #[error("{0}")] External(String),  // exit 2
}
impl QuarryError { pub fn exit_code(&self) -> u8 { match self { Self::Refusal(_) => 1, Self::External(_) => 2 } } }
impl From<std::io::Error> for QuarryError { fn from(e: std::io::Error) -> Self { Self::External(e.to_string()) } }
// same for rusqlite::Error, serde_json::Error

fn main() -> std::process::ExitCode {
    let cli = Cli::parse();
    match Context::build(&cli).and_then(|ctx| cli::run(&ctx, &cli)) {
        Ok(rendered) => { print!("{rendered}"); ExitCode::SUCCESS }
        Err(e) => {
            if cli.json { println!("{}", output::error_envelope(&e)) } else { eprintln!("{e}") }
            ExitCode::from(e.exit_code())
        }
    }
}
```

### docsrepo: the S1 retry loop

```rust
pub(crate) enum Redo { Rebuilt, Skip(UpdateResult) }

pub(crate) fn commit_and_push(ctx: &Context, message: &str, mut redo: impl FnMut() -> Result<Redo>) -> Result<PushResult> {
    let git = ctx.clone_git();
    for _attempt in 1..=3 {
        git.run(&["add", "-A"])?;
        git.run(&["commit", "-q", "-m", message])?;
        let out = git.run_unchecked(&["push", "origin", "HEAD:main"])?;
        if out.status.success() { return Ok(PushResult::Pushed); }
        let stderr = String::from_utf8_lossy(&out.stderr);
        if !(stderr.contains("rejected") || stderr.contains("fetch first")) {
            refresh(ctx)?;
            return Err(QuarryError::External(format!("push failed: {}", stderr.trim())));
        }
        refresh(ctx)?;
        if let Redo::Skip(r) = redo()? { return Ok(PushResult::Skipped(r)); }
    }
    Err(QuarryError::External("docs repo busy, retry".into()))
}
```

`refresh`: `fetch --depth 1 origin main`, then `status --porcelain` and `rev-list origin/main..HEAD --count` for the S11 "discarded …" line, then `reset --hard origin/main`. `write_folder`: `fs::remove_dir_all(clone/<repo>)` (ignore NotFound) then `fs::rename(tmp.into_path(), clone/<repo>)`; the `TempDir` is created under `.quarry/` so the rename stays on one filesystem.

```rust
pub(crate) fn build(ctx: &Context, sha: &Sha) -> Result<TempDir> {
    let tmp = tempfile::Builder::new().prefix(".build-").tempdir_in(ctx.quarry_dir())?;
    let tar = ctx.repo_git().run(&["archive", "--format=tar", &format!("{sha}:{}", ctx.config.docs_dir)])?;
    unpack_tar(&tar.stdout, tmp.path())?;                     // hand-rolled ustar reader: name, size, type; refuses `..`, absolute, links (Q4)
    let tree: HashSet<String> = ctx.repo_git().run(&["ls-tree", "-r", "--name-only", sha])?.stdout_lines();
    for md in walk_md(tmp.path()) {
        let text = fs::read_to_string(&md)?;
        let (fm, body) = split_frontmatter(&text);           // frontmatter untouched
        fs::write(&md, format!("{fm}{}", rewrite(body, &tree, &template, sha)))?;  // skips fenced blocks, leaves paths not in `tree`
    }
    fs::write(tmp.path().join(".quarry-stamp"), stamp_json(sha, &ctx.identity()?.origin))?;   // sorted keys + '\n'
    Ok(tmp)
}
```

### query: deps BFS

```rust
pub(crate) fn deps(index: &Index, repo: &RepoName, dir: Direction, depth: u32) -> Result<DepsResult> {
    let mut seen = HashSet::from([repo.clone()]);
    let mut frontier = VecDeque::from([(repo.clone(), 0u32)]);
    let mut out = Vec::new(); let mut truncated = false;
    while let Some((node, d)) = frontier.pop_front() {
        if depth != 0 && d >= depth { truncated = true; continue; }
        for e in index.edges(&node, dir)? {
            let other = e.other(dir);
            out.push(DepEdge { repo: other.clone(), edge: e.clone(), depth: d + 1, cycle: seen.contains(&other) });
            if seen.insert(other.clone()) && !e.missing { frontier.push_back((other, d + 1)); }
        }
    }
    Ok(DepsResult { edges: out, truncated })
}
```
